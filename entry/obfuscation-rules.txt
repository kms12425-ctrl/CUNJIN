-enable-filename-obfuscation
-enable-export-obfuscation

# ---------------------------------------------------------------------------
# Data-contract safety (added with the JSON persistence upgrade).
#
# cards_v2 / settings_v2 / work_state_v2 store JSON whose keys are exactly the
# property names of TaskCardData / CardSettings / WorkState.
# With -enable-property-obfuscation those names get renamed in release builds,
# and the rename mapping can differ between releases. That would make JSON
# written by one version unreadable by the next -> data loss on app update.
#
# These names are a persisted data contract, not implementation detail, so they
# must never be obfuscated.
# ---------------------------------------------------------------------------
-keep-property-name
schemaVersion
id
name
desc
length
timeLimitDays
targetHours
doneHours
startAt
endAt
punchCard
useBackdrop
colorSchemeLight
colorSchemeDark
blurRadius
minCardSize
appOpenTime
workingCardName
workStart
workLastEnd
